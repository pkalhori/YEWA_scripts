##the output .txt file from genmap was modified slightly to remove spaces between each position

input_file = "/home/pkalhori/msmc/mappability_mask.txt"
output_file = "/home/pkalhori/msmc/mappability_mask_modified.fa"

with open(input_file, "r") as infile, open(output_file, "wb") as outfile:
    for line in infile:
        # Preserve headers (lines starting with '>')
        if line.startswith(">"):
            outfile.write(line.encode())  # Write the header as-is in binary
        else:
            # Remove spaces and convert the line to binary
            binary_line = line.replace(" ", "").strip().encode()
            outfile.write(binary_line + b"\n")
